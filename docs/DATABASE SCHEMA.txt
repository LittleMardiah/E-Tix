-- Role pengguna dalam sistem
CREATE TYPE user_role_enum AS ENUM ('customer', 'event_organizer', 'admin', 'gate_staff');

-- Status akun pengguna
CREATE TYPE user_status_enum AS ENUM ('ACTIVE', 'SUSPENDED', 'FROZEN', 'DELETED');

-- Status tiket
CREATE TYPE ticket_status_enum AS ENUM (
  'AVAILABLE',           -- Kursi kosong, belum dibeli
  'LOCKED',              -- Sedang dalam proses checkout (TTL 3 menit)
  'PENDING_SPLIT_PAY',   -- Menunggu pembayaran grup
  'CONFIRMED',           -- Sudah dibayar, tiket valid
  'RESALE_PENDING',      -- Sedang dalam proses jual-beli
  'RESOLD',              -- Tiket lama sudah dijual kembali (audit trail)
  'CANCELLED',           -- Dibatalkan (refund sudah diproses)
  'USED'                 -- Sudah di-scan di gate
);

-- Status transaksi pembayaran
CREATE TYPE payment_status_enum AS ENUM ('PENDING', 'PAID', 'EXPIRED', 'FAILED', 'REFUNDED', 'CANCELLED');

-- Mode pembayaran grup
CREATE TYPE payment_mode_enum AS ENUM ('SPLIT', 'BULK');

-- Metode pembayaran
CREATE TYPE payment_method_enum AS ENUM ('VA', 'E_WALLET', 'CREDIT_CARD');

-- Status transaksi resale
CREATE TYPE resale_status_enum AS ENUM ('PENDING', 'COMPLETED', 'CANCELLED', 'PENDING_PAYOUT');

-- Status scan di gate
CREATE TYPE scan_status_enum AS ENUM ('VALID', 'INVALID', 'EXPIRED', 'ALREADY_USED');

-- Action type untuk admin log
CREATE TYPE admin_action_type_enum AS ENUM (
  'BULK_CANCEL',
  'MANUAL_PAID',
  'MANUAL_REFUND',
  'FORCE_RELEASE_SEAT',
  'UPDATE_EVENT_PRICE',
  'SUSPEND_USER',
  'UNSUSPEND_USER',
  'BLACKLIST_IP',
  'REMOVE_BLACKLIST_IP'
);

-- Status event
CREATE TYPE event_status_enum AS ENUM ('DRAFT', 'PUBLISHED', 'ON_SALE', 'SOLD_OUT', 'COMPLETED', 'CANCELLED');

-- Jenis insight dari feedback loop
CREATE TYPE insight_type_enum AS ENUM ('GATE_STAFFING', 'PRICE_RECOMMENDATION', 'BOT_BLACKLIST', 'PEAK_TIME');

CREATE TABLE IF NOT EXISTS users (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  email VARCHAR(255) UNIQUE NOT NULL,
  phone VARCHAR(20) UNIQUE NOT NULL,
  name VARCHAR(255) NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  role user_role_enum NOT NULL DEFAULT 'customer',
  status user_status_enum NOT NULL DEFAULT 'ACTIVE',

  updated_at TIMESTAMPTZ DEFAULT NOW(),

  CONSTRAINT phone_valid CHECK (phone ~* '^\+?[0-9]{10,15}$'),

-- ============================================================
-- TRIGGER 1: Update booking status when all obligations paid
-- ============================================================

CREATE OR REPLACE FUNCTION check_booking_payment_complete()
RETURNS TRIGGER AS $$
DECLARE
  booking_record bookings%ROWTYPE;
  seat_pos INTEGER;
  obligation_record payment_obligations%ROWTYPE;
BEGIN
  IF NEW.status = 'PAID' THEN
    SELECT * INTO booking_record FROM bookings WHERE id = NEW.booking_id FOR UPDATE;
    
    IF booking_record.payment_status = 'PENDING' THEN
      IF NOT EXISTS (
        SELECT 1 FROM payment_obligations 
        WHERE booking_id = NEW.booking_id 
        AND status != 'PAID'
      ) THEN
        UPDATE bookings 
        SET payment_status = 'PAID', updated_at = NOW()
        WHERE id = NEW.booking_id;
        
        IF booking_record.payment_mode = 'BULK' THEN
          FOR seat_pos IN SELECT jsonb_array_elements_text(booking_record.seat_positions)::integer
          LOOP
            INSERT INTO tickets (
              booking_id,
              payment_obligation_id,
              owner_user_id,
              section_id,
              seat_position,
              qr_secret_key,
              fallback_pin,
              status,
              is_assigned
            ) VALUES (
              booking_record.id,
              NEW.id,
              booking_record.leader_user_id,
              booking_record.section_id,
              seat_pos,
              gen_random_uuid()::text,
              UPPER(SUBSTRING(REPLACE(gen_random_uuid()::text, '-', ''), 1, 12)),
              'CONFIRMED',
              FALSE
            );
          END LOOP;
          
        ELSE
          -- SPLIT mode: validate seat_position is NOT NULL
          FOR obligation_record IN 
            SELECT * FROM payment_obligations 
            WHERE booking_id = NEW.booking_id 
            AND status = 'PAID'
          LOOP
            IF obligation_record.seat_position IS NULL THEN
              RAISE EXCEPTION 'seat_position cannot be NULL for SPLIT mode payment obligation %', obligation_record.id;
            END IF;
            
            INSERT INTO tickets (
              booking_id,
              payment_obligation_id,
              owner_user_id,
              section_id,
              seat_position,
              qr_secret_key,
              fallback_pin,
              status,
              is_assigned
            ) VALUES (
              booking_record.id,
              obligation_record.id,
              obligation_record.user_id,
              booking_record.section_id,
              obligation_record.seat_position,
              gen_random_uuid()::text,
              UPPER(SUBSTRING(REPLACE(gen_random_uuid()::text, '-', ''), 1, 12)),
              'CONFIRMED',
              TRUE
            );
          END LOOP;
        END IF;
      END IF;
    END IF;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ============================================================
-- TRIGGER 4: Auto-generate Post-Event Report (F004b)
-- ============================================================

CREATE OR REPLACE FUNCTION generate_post_event_report()
RETURNS TRIGGER AS $$
DECLARE
  report_data JSONB;
  gate_utilization JSONB;
BEGIN
  IF NEW.status = 'COMPLETED' THEN
    WITH gate_stats AS (
      SELECT 
        ts.gate_id,
        COUNT(*) as total_scans,
        SUM(CASE WHEN ts.scan_status = 'VALID' THEN 1 ELSE 0 END) as valid_scans
      FROM ticket_scans ts
      JOIN tickets t ON t.id = ts.ticket_id
      JOIN bookings b ON b.id = t.booking_id
      WHERE b.event_id = NEW.id
      GROUP BY ts.gate_id
    ),
    gate_agg AS (
      SELECT jsonb_object_agg(
        gate_id, 
        jsonb_build_object('total', total_scans, 'valid', valid_scans)
      ) as gate_utilization
      FROM gate_stats
    ),
    ticket_stats AS (
      SELECT 
        COUNT(*) as total_tickets,
        SUM(CASE WHEN status = 'USED' THEN 1 ELSE 0 END) as used_tickets,
        SUM(CASE WHEN status = 'CONFIRMED' THEN 1 ELSE 0 END) as unused_tickets
      FROM tickets t
      JOIN bookings b ON b.id = t.booking_id
      WHERE b.event_id = NEW.id
    ),
    loyal_fans AS (
      SELECT COUNT(DISTINCT u.id) as loyal_customers
      FROM users u
      JOIN tickets t ON t.owner_user_id = u.id
      JOIN bookings b ON b.id = t.booking_id
      WHERE b.event_id = NEW.id
      AND u.total_purchases > 1
    )
    SELECT 
      jsonb_build_object(
        'event_id', NEW.id,
        'event_name', NEW.name,
        'total_tickets', (SELECT total_tickets FROM ticket_stats),
        'used_tickets', (SELECT used_tickets FROM ticket_stats),
        'unused_tickets', (SELECT unused_tickets FROM ticket_stats),
        'no_show_rate', 
          CASE 
            WHEN (SELECT total_tickets FROM ticket_stats) > 0 
            THEN ROUND(((SELECT unused_tickets FROM ticket_stats)::NUMERIC / (SELECT total_tickets FROM ticket_stats)::NUMERIC) * 100, 2)
            ELSE 0
          END,
        'total_scans', COALESCE((SELECT SUM(total_scans) FROM gate_stats), 0),
        'valid_scans', COALESCE((SELECT SUM(valid_scans) FROM gate_stats), 0),
        'gate_utilization', COALESCE((SELECT gate_utilization FROM gate_agg), '{}'::jsonb),
        'loyal_customers', (SELECT loyal_customers FROM loyal_fans),
        'generated_at', NOW()
      ) INTO report_data;
    
    INSERT INTO insights (
      event_id,
      insight_type,
      title,
      description,
      recommendation
    ) VALUES (
      NEW.id,
      'PEAK_TIME',
      'Post-Event Report - ' || NEW.name,
      'Auto-generated report after event completion',
      report_data
    );
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_generate_post_event_report
AFTER UPDATE OF status ON events
FOR EACH ROW
WHEN (NEW.status = 'COMPLETED' AND OLD.status != 'COMPLETED')
EXECUTE FUNCTION generate_post_event_report();

-- ============================================================
-- TRIGGER 5: Handle Resale Lifecycle (INSERT + UPDATE) (F005)
-- ============================================================

CREATE OR REPLACE FUNCTION handle_resale_lifecycle()
RETURNS TRIGGER AS $$
DECLARE
  old_ticket_record tickets%ROWTYPE;
  new_ticket_id UUID;
BEGIN
  -- CASE 1: New resale listing (INSERT)
  IF TG_OP = 'INSERT' THEN
    UPDATE tickets 
    SET status = 'RESALE_PENDING', updated_at = NOW()
    WHERE id = NEW.ticket_id;
    
    RETURN NEW;
  END IF;
  
  -- CASE 2: Resale completed (UPDATE to COMPLETED)
  IF TG_OP = 'UPDATE' AND NEW.status = 'COMPLETED' AND OLD.status != 'COMPLETED' THEN
    -- Get current ticket details
    SELECT * INTO old_ticket_record FROM tickets WHERE id = NEW.ticket_id;
    
    -- Step 1: Mark old ticket as RESOLD (audit trail)
    UPDATE tickets 
    SET 
      status = 'RESOLD',
      original_owner_user_id = OLD.seller_user_id,
      resale_transaction_id = NEW.id,
      updated_at = NOW()
    WHERE id = NEW.ticket_id;
    
    -- Step 2: Create new ticket for buyer
    INSERT INTO tickets (
      booking_id,
      payment_obligation_id,
      owner_user_id,
      section_id,
      seat_position,
      status,
      is_assigned,
      qr_secret_key,
      fallback_pin,
      original_owner_user_id,
      resale_transaction_id,
      created_at
    ) VALUES (
      old_ticket_record.booking_id,
      old_ticket_record.payment_obligation_id,
      NEW.buyer_user_id,
      old_ticket_record.section_id,
      old_ticket_record.seat_position,
      'CONFIRMED',
      TRUE,
      gen_random_uuid()::text,
      UPPER(SUBSTRING(REPLACE(gen_random_uuid()::text, '-', ''), 1, 12)),
      OLD.seller_user_id,
      NEW.id,
      NOW()
    ) RETURNING id INTO new_ticket_id;
    
    RETURN NEW;
  END IF;
  
  -- CASE 3: Resale cancelled (UPDATE to CANCELLED)
  IF TG_OP = 'UPDATE' AND NEW.status = 'CANCELLED' AND OLD.status != 'CANCELLED' THEN
    -- Return ticket to CONFIRMED status (owner remains seller)
    UPDATE tickets 
    SET 
      status = 'CONFIRMED',
      resale_transaction_id = NULL,
      updated_at = NOW()
    WHERE id = NEW.ticket_id;
    
    RETURN NEW;
  END IF;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_handle_resale_lifecycle
AFTER INSERT OR UPDATE OF status ON resale_transactions
FOR EACH ROW
EXECUTE FUNCTION handle_resale_lifecycle();

RETURNS TRIGGER AS $$
BEGIN
  IF NEW.status IN ('REFUNDED', 'FAILED') AND OLD.status NOT IN ('REFUNDED', 'FAILED') THEN
    UPDATE tickets 
    SET status = 'CANCELLED', updated_at = NOW()
    WHERE payment_obligation_id = NEW.id
    AND status NOT IN ('CANCELLED', 'USED', 'RESOLD');
  END IF;
  
  RETURN NEW;
END;

  END as status

  DATE(scanned_at) as scan_date,
  EXTRACT(HOUR FROM scanned_at) as hour,
  COUNT(*) as total_scans,
  SUM(CASE WHEN scan_status = 'VALID' THEN 1 ELSE 0 END) as valid_scans
FROM ticket_scans
WHERE scanned_at > NOW() - INTERVAL '24 hours'
GROUP BY gate_id, scan_date, hour
ORDER BY hour DESC;

  name,
  total_purchases,
  base_score,
  CASE 
    WHEN (base_score + total_purchases * 10) >= 90 THEN 'TIER_1'
    WHEN (base_score + total_purchases * 10) >= 70 THEN 'TIER_2'

  END as fan_tier

ORDER BY (base_score + total_purchases * 10) DESC;
