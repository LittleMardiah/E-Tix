1. EXECUTIVE SUMMARY
================================================================================

PROBLEM BEING SOLVED
E-Tix Flash solves 3 critical problems in ticketing:
1. Server crashes during flash sales (20.000+ RPS)
2. Scalper bots buying tickets in seconds
3. Groups failing to secure adjacent seats due to race conditions

MVP SCOPE (V1.0 — ALL FEATURES IN)
| Feature ID | Feature Name | Priority |
|------------|-------------|----------|
| F001 | Behavioral Ingress & Fan-Score Queueing (with Priority Tier) | P0 |
| F002 | Bitmask Multi-Seat Atomic Locking (Partitioned per Section) | P0 |
| F003 | Group Split-Pay Execution Engine (Hybrid: Split + Bulk) | P0 |
| F004 | Dynamic HMAC QR + Offline Fallback PIN | P0 |
| F004b | Gate Analytics & Post-Event Intelligence (Feedback Loop) | P0 |
| F005 | Controlled Secondary Resale Market | P0 |
| F006 | Admin Emergency & Reconciliation Tools | P0 |

*Semua fitur adalah P0. Tidak ada fitur yang ditunda.*

SUCCESS METRICS
| Metric | Target | Validation |
|--------|--------|------------|
| API Latency (p95) | < 100 ms | Prometheus + Grafana |
| Bot Filtering Rate | > 95% | Load testing + bot scripts |
| Group Lock Success Rate | 100% (No Race Condition) | Concurrent test (10 drivers → 1 winner) |
| System Uptime | 99.9% | During flash sale period |
| Post-Event Report Generation | < 1 hour | After event ends |

ARCHITECTURE DECISION
- Backend: Golang 1.22+ (Modular Monolith — 1 binary)
- Frontend: Next.js 14+ (PWA — 1 codebase for all roles)
- Database: Supabase PostgreSQL 15+
- Cache: Redis (Upstash)
- Message Broker: RabbitMQ (CloudAMQP)
- Deployment: Render.com (backend) + Vercel (frontend)

CONNECTION POOL CONFIGURATION

go
import "github.com/jackc/pgx/v5/pgxpool"

func NewDBPool(ctx context.Context, connString string) (*pgxpool.Pool, error) {
    config, err := pgxpool.ParseConfig(connString)
    if err != nil {
        return nil, err
    }
    
    config.MaxConns = 15                     // Supabase free tier
    config.MinConns = 2
    config.MaxConnIdleTime = 5 * time.Minute
    config.MaxConnLifetime = 30 * time.Minute
    config.HealthCheckPeriod = 1 * time.Minute
    
    // CRITICAL: Set statement timeout for all queries
    config.ConnConfig.RuntimeParams["statement_timeout"] = "3000ms"
    
    // Use simple protocol for better performance
    config.ConnConfig.DefaultQueryExecMode = pgx.QueryExecModeSimpleProtocol
    
    return pgxpool.NewWithConfig(ctx, config)
}
DEADLOCK PREVENTION (ORDER BY id ASC)

go
// WAJIB: Semua transaksi multi-row menggunakan ORDER BY id ASC
func (r *BookingRepository) ProcessSplitPayment(ctx context.Context, bookingID uuid.UUID) error {
    tx, err := r.db.BeginTx(ctx, nil)
    if err != nil {
        return err
    }
    defer tx.Rollback()
    
    // Lock booking
    var booking Booking
    err = tx.QueryRowContext(ctx, `
        SELECT id, payment_status, total_seats, seat_positions
        FROM bookings WHERE id = $1 FOR UPDATE
    `, bookingID).Scan(&booking.ID, &booking.PaymentStatus, &booking.TotalSeats, &booking.SeatPositions)
    if err != nil {
        return err
    }
    
    // Lock payment obligations with ORDER BY id ASC (CRITICAL)
    rows, err := tx.QueryContext(ctx, `
        SELECT id, user_id, amount, seat_position
        FROM payment_obligations
        WHERE booking_id = $1
        ORDER BY id ASC
        FOR UPDATE
    `, bookingID)
    if err != nil {
        return err
    }
    defer rows.Close()
    
    // Process...
    
    return tx.Commit()
}

6.1 F001 — BEHAVIORAL INGRESS (2-STAGE PIPELINE)

Stage 1 (Synchronous — Target < 1ms):

go
func (m *IngressMiddleware) Handle(c *gin.Context) {
    userID := c.GetHeader("X-User-ID")
    ip := c.ClientIP()
    
    // Rate Limiter (Token Bucket)
    if !m.rateLimiter.Allow(ip, "ingress", 100, time.Minute) {
        c.JSON(429, ErrorResponse("RATE_LIMIT_EXCEEDED"))
        return
    }
    
    // Blacklist check (memory cache, updated from Redis every 60s)
    if m.blacklist.IsBlacklisted(ip) {
        c.JSON(403, ErrorResponse("BLACKLISTED"))
        return
    }
    
    // Base Score from Redis
    baseScore, _ := m.redis.Get(ctx, "user:score:"+userID+":base").Int()
    if baseScore == 0 {
        baseScore = 50 // default for new users
    }
    
    // Dynamic Score from Redis
    dynamicScore, _ := m.redis.Get(ctx, "user:score:"+userID+":dynamic").Int()
    
    finalScore := baseScore + dynamicScore
    
    // Priority Tier
    var tier string
    switch {
    case finalScore > 90:
        tier = "TIER_1"
    case finalScore >= 70:
        tier = "TIER_2"
    default:
        tier = "TIER_3"
    }
    
    // Generate Queue Token (JWT with jti)
    token := generateQueueToken(userID, tier, finalScore)

func (s *BookingService) LockSeats(ctx context.Context, req *LockRequest) (*LockResponse, error) {
    // Validate Queue Token (single-use)
    if !s.validateQueueToken(req.Token) {
        return nil, ErrTokenAlreadyUsed
    }
    
    // Get layout_version from database
    layoutVersion, err := s.repo.GetSectionLayoutVersion(ctx, req.SectionID)
    if err != nil {
        return nil, err
    }
    
    key := fmt.Sprintf("seats:%s:%s:v%d", req.EventID, req.SectionID, layoutVersion)
    args := []string{
        jsonEncode(req.SeatPositions),
        req.BookingID.String(),
        jsonEncode(req.BookingData),
    }
    
    var result interface{}
    var err error
    
    result, err = s.circuitBreaker.Execute(func() (interface{}, error) {
        return s.redis.Eval(ctx, lockScript, []string{key}, args).Result()
    })
    
    if err != nil {
        if errors.Is(err, gobreaker.ErrOpenState) {
            return s.fallbackLockSeats(ctx, req)
        }
        return nil, err
    }
    
    var resp LockResponse
    json.Unmarshal([]byte(result.(string)), &resp)
    
    if resp.Status == "CONFLICT" {
        return nil, ErrSeatsNotAvailable
    }
    
    booking := &Booking{
        ID:             req.BookingID,
        EventID:        req.EventID,
        SectionID:      req.SectionID,
        LeaderUserID:   req.UserID,
        SeatPositions:  req.SeatPositions,
        TotalSeats:     len(req.SeatPositions),
        TotalAmount:    req.TotalAmount,
        PaymentMode:    req.PaymentMode,
        PaymentStatus:  "PENDING",
        ExpiresAt:      time.Now().Add(180 * time.Second),
        LayoutVersion:  layoutVersion,
    }
    
    err = s.repo.CreateBooking(ctx, booking)
    if err != nil {
        s.releaseSeats(ctx, req.EventID, req.SectionID, req.BookingID, req.SeatPositions, layoutVersion)
        return nil, err
    }
    
    s.queue.PublishDelayed("release_seat_check", booking.ID, 180*time.Second)
    
    return &LockResponse{BookingID: booking.ID, ExpiresAt: booking.ExpiresAt}, nil
}

func VerifyDynamicQR(qrToken string, secretKey string) (bool, uuid.UUID, error) {
    data, err := base64.URLEncoding.DecodeString(qrToken)
    if err != nil {
        return false, uuid.Nil, err
    }
    
    parts := strings.Split(string(data), "|")
    if len(parts) != 2 {
        return false, uuid.Nil, errors.New("invalid QR format")
    }
    
    ticketID, err := uuid.Parse(parts[0])
    if err != nil {
        return false, uuid.Nil, err
    }
    
    signature := parts[1]
    
    currentBlock := time.Now().Unix() / 30
    for _, block := range []int64{currentBlock - 1, currentBlock, currentBlock + 1} {
        payload := fmt.Sprintf("%s|%d", ticketID.String(), block)
        mac := hmac.New(sha256.New, []byte(secretKey))
        mac.Write([]byte(payload))
        expected := base64.URLEncoding.EncodeToString(mac.Sum(nil))
        
        if hmac.Equal([]byte(signature), []byte(expected)) {
            return true, ticketID, nil
        }
    }
    
    return false, uuid.Nil, errors.New("invalid signature or expired")
}

    logger *zap.Logger
}

func (w *ResaleExpiryWorker) Process(ctx context.Context, msg []byte) error {
    var payload model.ResaleExpiryPayload
    if err := json.Unmarshal(msg, &payload); err != nil {
        w.logger.Error("failed to unmarshal resale expiry payload", zap.Error(err))
        return err
    }
    
    resaleID := payload.ResaleID
    
    tx, err := w.repo.BeginTx(ctx)
    if err != nil {
        return err
    }
    defer tx.Rollback()
    
    var existingID string
    err = tx.QueryRowContext(ctx, `
        SELECT id FROM resale_transactions
        WHERE id = $1 AND status = 'RESERVED'
        FOR UPDATE
    `, resaleID).Scan(&existingID)
    if err != nil {
        return nil
    }
    
    _, err = tx.ExecContext(ctx, `
        UPDATE resale_transactions
        SET 
            buyer_user_id = NULL,
            status = 'PENDING',
            reservation_expires_at = NULL,
            updated_at = NOW()
        WHERE id = $1
    `, resaleID)
    if err != nil {
        return err
    }
    
    _, err = tx.ExecContext(ctx, `
        INSERT INTO admin_actions (
            admin_id, action_type, target_resource_type,
            target_resource_id, description, reason
        ) VALUES (
            $1,
            'FORCE_RELEASE_SEAT',
            'resale_transaction',
            $2,
            'Resale reservation expired',
            'RESERVATION_EXPIRED'
        )
    `, model.SystemUserUUID, resaleID)
    if err != nil {
        return err
    }
    
    return tx.Commit()
}

// internal/worker/db_queue_sync.go
package worker

import (
    "context"
    "time"
    
    "github.com/redis/go-redis/v9"
    "go.uber.org/zap"
)

type DBQueueSyncWorker struct {
    repo     *booking.Repository
    redis    *redis.Client
    circuit  *gobreaker.CircuitBreaker
    logger   *zap.Logger
}

func (w *DBQueueSyncWorker) Run(ctx context.Context) {
    ticker := time.NewTicker(30 * time.Second)
    defer ticker.Stop()
    
    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            w.processQueuedBookings(ctx)
        }
    }
}

func (w *DBQueueSyncWorker) processQueuedBookings(ctx context.Context) {
    // Check if circuit breaker is CLOSED (Redis is healthy)
    if w.circuit.State() != gobreaker.StateClosed {
        return // Still in recovery, wait
    }
    
    // Fetch queued bookings from PostgreSQL
    queuedBookings, err := w.repo.GetQueuedBookings(ctx, 100) // batch size
    if err != nil {
        w.logger.Error("failed to get queued bookings", zap.Error(err))
        return
    }
    
    if len(queuedBookings) == 0 {
        return
    }
    
    for _, booking := range queuedBookings {
        // Re-process each booking back to Redis
        err := w.repo.ProcessQueuedBooking(ctx, booking)
        if err != nil {
            w.logger.Error("failed to process queued booking", 
                zap.String("booking_id", booking.ID.String()),
                zap.Error(err),
            )
            continue
        }
        
        // Mark as processed
        err = w.repo.MarkBookingProcessed(ctx, booking.ID)
        if err != nil {
            w.logger.Error("failed to mark booking as processed",
                zap.String("booking_id", booking.ID.String()),
                zap.Error(err),
            )
        }
    }
    
    w.logger.Info("processed queued bookings", 
        zap.Int("count", len(queuedBookings)),
    )
}

func (h *AuthHandler) RefreshToken(c *gin.Context) {
    var req RefreshTokenRequest
    if err := c.BindJSON(&req); err != nil {
        c.JSON(400, ErrorResponse("INVALID_REQUEST"))
        return
    }
    
    claims, err := h.jwtService.ParseRefreshToken(req.RefreshToken)
    if err != nil {
        c.JSON(401, ErrorResponse("UNAUTHORIZED"))
        return
    }
    
    if claims.TokenType != "refresh" {
        c.JSON(401, ErrorResponse("TOKEN_TYPE_MISMATCH"))
        return
    }
    
    blacklisted, _ := h.redis.Get(ctx, "refresh_blacklist:"+claims.JTI).Result()
    if blacklisted == "1" {
        c.JSON(401, ErrorResponse("TOKEN_BLACKLISTED"))
        return
    }
    
    user, err := h.repo.GetUserByID(ctx, claims.UserID)
    if err != nil {
        c.JSON(404, ErrorResponse("NOT_FOUND"))
        return
    }
    
    if user.Status != "ACTIVE" {
        c.JSON(403, ErrorResponse("FORBIDDEN"))
        return
    }
    
    ttl := time.Until(time.Unix(claims.Exp, 0))
    if ttl <= 0 {
        ttl = 1 * time.Second
    }
    h.redis.Set(ctx, "refresh_blacklist:"+claims.JTI, "1", ttl)
    
    newAccessToken, err := h.jwtService.GenerateAccessToken(user)
    if err != nil {
        c.JSON(500, ErrorResponse("INTERNAL_ERROR"))
        return
    }
    
    newRefreshToken, err := h.jwtService.GenerateRefreshToken(user)
    if err != nil {
        c.JSON(500, ErrorResponse("INTERNAL_ERROR"))
        return
    }
    
    c.JSON(200, gin.H{
        "data": gin.H{
            "access_token":  newAccessToken,
            "refresh_token": newRefreshToken,
            "expires_in":    3600,
        },
    })
}
